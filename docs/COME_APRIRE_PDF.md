### Metodo 2: Usando Node.js + puppeteer

```bash
# Installa puppeteer (se non presente)
npm install puppeteer

# Creo script di conversione
cat > convert_to_pdf.js << 'EOF'
const puppeteer = require('puppeteer');

(async () => {
  const browser = await puppeteer.launch();
  const page = await browser.newPage();
  
  await page.goto('file://C:\\Claude\\tennistavolo-booking\\docs\\ANALISI_FILE_MANAGEMENT.html', {
    waitUntil: 'networkidle0'
  });
  
  await page.pdf({
    path: 'C:\\Claude\\tennistavolo-booking\\docs\\ANALISI_FILE_MANAGEMENT.pdf',
    format: 'A4',
    printBackground: true,
    margin: { top: '20px', bottom: '20px', left: '20px', right: '20px' }
  });
  
  await browser.close();
  console.log('PDF creato: ANALISI_FILE_MANAGEMENT.pdf');
})();
EOF

# Esegui conversione
node convert_to_pdf.js
```

### Metodo 3: Python + Selenium + pdfkit

```bash
# Installa dipendenze
pip install pdfkit selenium

# Creo script
cat > convert_to_pdf.py << 'EOF'
import pdfkit

options = {
    'page-size': 'A4',
    'margin-top': '0.75in',
    'margin-right': '0.75in',
    'margin-bottom': '0.75in',
    'margin-left': '0.75in',
    'print-media-type': None
}

pdfkit.from_file(
    'C:\\Claude\\tennistavolo-booking\\docs\\ANALISI_FILE_MANAGEMENT.html',
    'C:\\Claude\\tennistavolo-booking\\docs\\ANALISI_FILE_MANAGEMENT.pdf',
    options=options
)

print("PDF creato!")
EOF

# Esegui
python convert_to_pdf.py
```
